-- ═══════════════════════════════════════════════════════════════ -- GIRO FÁCIL · Autor da observação (rodar uma vez no SQL Editor) -- 1) cria as colunas que guardam quem escreveu -- 2) preenche as observações que já existem, pelo nome escrito no -- começo do texto ("Andressa - ...", "Ana: ...", "Marcos- ...") -- O texto das observações NÃO é alterado. -- ═══════════════════════════════════════════════════════════════ ALTER TABLE fgf_propostas ADD COLUMN IF NOT EXISTS observacao_por text; ALTER TABLE fgf_propostas ADD COLUMN IF NOT EXISTS observacao_anterior_por text; -- Nome no começo do texto → colaborador do Giro Fácil (ativo ou não). -- Aceita o nome completo ("ANA PAULA") ou só o primeiro nome ("Ana"). WITH nomes AS ( SELECT nome, upper(translate(nome, 'ÁÀÂÃÉÊÍÓÔÕÚÇáàâãéêíóôõúç', 'AAAAEEIOOOUCaaaaeeiooouc')) AS n, split_part(upper(translate(nome, 'ÁÀÂÃÉÊÍÓÔÕÚÇáàâãéêíóôõúç', 'AAAAEEIOOOUCaaaaeeiooouc')), ' ', 1) AS primeiro FROM config_colaboradores WHERE sistema = 'fgf' ), -- primeiro nome só vale se não houver duas pessoas com o mesmo unicos AS ( SELECT nome, n, primeiro FROM nomes WHERE primeiro IN (SELECT primeiro FROM nomes GROUP BY primeiro HAVING count(*) = 1) OR n = primeiro ), achados AS ( SELECT p.codigo, u.nome FROM fgf_propostas p JOIN unicos u ON upper(translate(p.observacao, 'ÁÀÂÃÉÊÍÓÔÕÚÇáàâãéêíóôõúç', 'AAAAEEIOOOUCaaaaeeiooouc')) ~ ('^\s*(' || u.n || '|' || u.primeiro || ')(\s+[A-Z]+)?\s*[-–—:]') WHERE p.observacao_por IS NULL AND coalesce(trim(p.observacao), '') <> '' ) UPDATE fgf_propostas p SET observacao_por = a.nome FROM achados a WHERE a.codigo = p.codigo; -- Mesma coisa para a observação anterior WITH nomes AS ( SELECT nome, upper(translate(nome, 'ÁÀÂÃÉÊÍÓÔÕÚÇáàâãéêíóôõúç', 'AAAAEEIOOOUCaaaaeeiooouc')) AS n, split_part(upper(translate(nome, 'ÁÀÂÃÉÊÍÓÔÕÚÇáàâãéêíóôõúç', 'AAAAEEIOOOUCaaaaeeiooouc')), ' ', 1) AS primeiro FROM config_colaboradores WHERE sistema = 'fgf' ), unicos AS ( SELECT nome, n, primeiro FROM nomes WHERE primeiro IN (SELECT primeiro FROM nomes GROUP BY primeiro HAVING count(*) = 1) OR n = primeiro ), achados AS ( SELECT p.codigo, u.nome FROM fgf_propostas p JOIN unicos u ON upper(translate(p.observacoes_anteriores, 'ÁÀÂÃÉÊÍÓÔÕÚÇáàâãéêíóôõúç', 'AAAAEEIOOOUCaaaaeeiooouc')) ~ ('^\s*(' || u.n || '|' || u.primeiro || ')(\s+[A-Z]+)?\s*[-–—:]') WHERE p.observacao_anterior_por IS NULL AND coalesce(trim(p.observacoes_anteriores), '') <> '' ) UPDATE fgf_propostas p SET observacao_anterior_por = a.nome FROM achados a WHERE a.codigo = p.codigo; -- Conferência: quantas observações ficaram com autor e quantas sem SELECT coalesce(observacao_por, '(sem nome no texto)') AS autor, count(*) AS observacoes FROM fgf_propostas WHERE coalesce(trim(observacao), '') <> '' GROUP BY 1 ORDER BY 2 DESC;
